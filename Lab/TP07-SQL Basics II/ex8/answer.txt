-- 1: products with buyPrice above average
SELECT productCode, productName, buyPrice
FROM products
WHERE buyPrice > (SELECT AVG(buyPrice) FROM products);

-- 2: customers in same country as 'Signal Gift Stores'
SELECT *
FROM customers
WHERE country = (
	SELECT country FROM customers WHERE customerName = 'Signal Gift Stores' LIMIT 1
);

-- 3: orders placed after the most recent order by customer 103
SELECT orderNumber, orderDate
FROM orders
WHERE orderDate > (
	SELECT MAX(orderDate) FROM orders WHERE customerNumber = 103
);
